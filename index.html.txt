      const [isAdmin, setIsAdmin] = useState(false);
      const [userTier, setUserTier] = useState("free"); // "free" | "premium"
      const [recipesToday, setRecipesToday] = useState(0);
      const [isPaywallOpen, setIsPaywallOpen] = useState(false);
      
      // Estado de Receta & Cocina
      const [ingredientsInput, setIngredientsInput] = useState("");
      const [selectedCategory, setSelectedCategory] = useState("Cualquiera");

      const [isGenerating, setIsGenerating] = useState(false);
      const [cookingStepIndex, setCookingStepIndex] = useState(null); // Modo paso a paso
      const [activeTimer, setActiveTimer] = useState(null); // { secondsLeft, isRunning }
      
      // Recetas Guardadas
      const [savedRecipes, setSavedRecipes] = useState([]);
      const [isSavedDrawerOpen, setIsSavedDrawerOpen] = useState(false);

      // Puerta Secreta Admin
      const [adminTapCount, setAdminTapCount] = useState([]);
      const [showSecretModal, setShowSecretModal] = useState(false);
      const [secretInputKey, setSecretInputKey] = useState("");
      const [secretError, setSecretError] = useState(null);
      const [showAdminToast, setShowAdminToast] = useState(false);

      // Carga inicial y persistencia
      useEffect(() => {
        try {
          const savedAdmin = localStorage.getItem("chefia_admin_key") === "true";
          setIsAdmin(savedAdmin);

          const savedCount = parseInt(localStorage.getItem("chefia_recipes_count") || "0", 10);
          const savedDate = localStorage.getItem("chefia_recipes_date");
          const todayStr = new Date().toDateString();

          if (savedDate === todayStr) {
            setRecipesToday(savedCount);
          } else {
            localStorage.setItem("chefia_recipes_date", todayStr);
            localStorage.setItem("chefia_recipes_count", "0");
            setRecipesToday(0);
          }

          const stored = localStorage.getItem("chefia_saved_recipes");
          if (stored) setSavedRecipes(JSON.parse(stored));
        } catch (e) {
          console.error("Storage error:", e);
        }

      // Temporizador del Modo Cocina

        let interval = null;
        if (activeTimer && activeTimer.isRunning && activeTimer.secondsLeft > 0) {
          interval = setInterval(() => {
            setActiveTimer(prev => ({
              ...prev,
              secondsLeft: prev.secondsLeft - 1
            }));
          }, 1000);
        } else if (activeTimer && activeTimer.secondsLeft === 0) {
          if (navigator.vibrate) navigator.vibrate([200, 100, 200]);
        }
        return () => clearInterval(interval);
      }, [activeTimer]);

      // Gesto Secreto: 3 toques en logo "C"
      const handleLogoClick = () => {
        const now = Date.now();
        const recent = [...adminTapCount.filter(t => now - t < 1500), now];
        setAdminTapCount(recent);

        if (recent.length >= 3) {
          setAdminTapCount([]);
          setSecretInputKey("");
          setSecretError(null);
          setShowSecretModal(true);
        }
      };

      // Validación de clave Admin "Unai7227"
      const handleSecretSubmit = (e) => {
        e.preventDefault();
        if (secretInputKey === "Unai7227") {
          localStorage.setItem("chefia_admin_key", "true");
          setIsAdmin(true);
          setIsPaywallOpen(false);
          setShowSecretModal(false);
          setShowAdminToast(true);
          setTimeout(() => setShowAdminToast(false), 4000);
        } else {
          setSecretError("Clave incorrecta");
        }
      };

      // Generación de Receta
      const handleGenerateRecipe = (customPromptText) => {
        const isUnlimited = isAdmin || userTier === "premium";
        
        // Bloqueo de Paywall para usuarios gratuitos (1 receta al día)
        if (!isUnlimited && recipesToday >= 1) {
          setIsPaywallOpen(true);
          return;
        }

        setIsGenerating(true);
        setCookingStepIndex(null);
        setActiveTimer(null);

        setTimeout(() => {
          // Seleccionar plantilla inteligente o adaptar según ingredientes
          let chosen = RECIPE_TEMPLATES[Math.floor(Math.random() * RECIPE_TEMPLATES.length)];
          const input = customPromptText || ingredientsInput;
          
          if (input.toLowerCase().includes("pollo")) chosen = RECIPE_TEMPLATES[1];
          else if (input.toLowerCase().includes("pasta") || input.toLowerCase().includes("champiñon")) chosen = RECIPE_TEMPLATES[2];
          else if (input.toLowerCase().includes("salmon") || input.toLowerCase().includes("pescado")) chosen = RECIPE_TEMPLATES[0];

          // Adaptar nombre si hay ingredientes custom
          let finalRecipe = { ...chosen, id: Date.now().toString() };
          if (input.trim().length > 0 && !customPromptText) {
            finalRecipe.description = `Receta diseñada a medida aprovechando: ${input.trim()}.`;
          }

          setCurrentRecipe(finalRecipe);
          setIsGenerating(false);

          // Actualizar contador diario
          if (!isUnlimited) {
            const nextCount = recipesToday + 1;
            setRecipesToday(nextCount);
            localStorage.setItem("chefia_recipes_count", nextCount.toString());
            localStorage.setItem("chefia_recipes_date", new Date().toDateString());
          }
        }, 1200);
      };

      // Guardar / Quitar Favoritos
      const toggleSaveRecipe = (recipe) => {
        let updated;
        const exists = savedRecipes.some(r => r.title === recipe.title);
        if (exists) {
          updated = savedRecipes.filter(r => r.title !== recipe.title);
        } else {
          updated = [recipe, ...savedRecipes];
        }
        setSavedRecipes(updated);
        localStorage.setItem("chefia_saved_recipes", JSON.stringify(updated));
      };

      // Desbloquear Pro
      const handleUnlockPro = () => {
        setUserTier("premium");
        setIsPaywallOpen(false);
      };

                    ✨ Generación inteligente de recetas
                  </span>
                  <h2 className="font-serif font-extrabold text-3xl sm:text-5xl text-[#2D2A26] tracking-tight leading-tight">
                    ¿Qué ingredientes tienes hoy en tu nevera?

                    Escribe lo que tengas a mano y la IA creará una receta gourmet paso a paso evitando el desperdicio.
                  </p>

                        Tipo de plato
                      </label>

                        value={selectedCategory}

                      >
                        <option>Cualquiera</option>
                        <option>Almuerzo nutritivo</option>
                        <option>Cena ligera</option>
                        <option>Vegetariano / Vegano</option>
                        <option>Alto en proteínas (Fitness)</option>
                      </select>

                        Tiempo disponible
                      </label>
                      <select 
                        value={selectedTime}
                        onChange={(e) => setSelectedTime(e.target.value)}
                        className="w-full p-3 rounded-xl bg-[#FBF9F5] border border-[#EBE7DF] text-xs font-semibold text-[#2D2A26] focus:outline-none focus:ring-1 focus:ring-[#FF6B35]"

                        <option>Rápido (&lt; 20 min)</option>
                        <option>Medio (20 - 40 min)</option>
                        <option>Elaborado (&gt; 40 min)</option>
                      </select>
                    </div>
                  </div>

                  {/* Botón Generar */}
                  <button
                    onClick={() => handleGenerateRecipe()}

                    className="w-full py-4 px-6 rounded-2xl bg-[#FF6B35] hover:bg-[#e05623] active:scale-[0.99] text-white font-extrabold text-base shadow-lg shadow-[#FF6B35]/25 transition-all flex items-center justify-center gap-2 cursor-pointer disabled:opacity-50"
                  >
                    {isGenerating ? (
                      <>
                        <svg className="animate-spin h-5 w-5 text-white" viewBox="0 0 24 24"><circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4" fill="none"></circle><path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8v8H4z"></path></svg>
                        <span>Creando tu receta personalizada...</span>

                      <>
                        <span>🍳 Generar Receta con IA</span>

                <div className="space-y-3">
                  <h3 className="text-xs font-bold uppercase tracking-wider text-[#8C8880] text-center">
                    O prueba con una idea rápida:

                    {[
                      "Salmón fresco con espárragos",
                      "Wok de pollo con verduras",
                      "Pasta cremosa con setas",
                      "Cena ligera en 15 minutos"
                    ].map((idea, idx) => (
                      <button
                        key={idx}
                        onClick={() => {
                          setIngredientsInput(idea);
                          handleGenerateRecipe(idea);
                        }}
                        className="px-3.5 py-2 rounded-xl bg-white border border-[#EBE7DF] text-xs font-semibold text-[#2D2A26] hover:border-[#FF6B35] hover:text-[#FF6B35] transition-all cursor-pointer shadow-xs"
                      >
                        ⚡ {idea}
                      </button>
                    ))}
                  </div>
                </div>
              </div>
            ) : (
              /* VISTA 2: DETALLE DE LA RECETA GENERADA */
              <div className="space-y-6 animate-fade-in">
                {/* Barra Superior con botón Volver y Guardar */}
                <div className="flex items-center justify-between">
                  <button
                    onClick={() => { setCurrentRecipe(null); setCookingStepIndex(null); }}
                    className="flex items-center gap-1.5 text-xs font-bold text-[#8C8880] hover:text-[#2D2A26] cursor-pointer"
                  >
                    ← Crear otra receta
                  </button>

                  <div className="flex items-center gap-2">
                    <button
                      onClick={() => toggleSaveRecipe(currentRecipe)}
                      className="px-3.5 py-1.5 rounded-full bg-white border border-[#EBE7DF] text-xs font-bold flex items-center gap-1.5 hover:bg-[#F2EFE9] transition-colors cursor-pointer"
                    >
                      <span>{savedRecipes.some(r => r.title === currentRecipe.title) ? "❤️ Guardada" : "🤍 Guardar"}</span>
                    </button>
                  </div>
                </div>

                {/* Tarjeta Principal de la Receta */}
                <div className="bg-white rounded-3xl p-6 sm:p-8 shadow-xl shadow-black/5 border border-[#EBE7DF] space-y-6">
                  {/* Cabecera */}
                  <div>
                    <div className="flex flex-wrap items-center gap-2 mb-2">
                      <span className="px-2.5 py-1 rounded-lg bg-[#FF6B35]/10 text-[#FF6B35] text-xs font-black uppercase tracking-wider">
                        {currentRecipe.difficulty}
                      </span>
                      <span className="px-2.5 py-1 rounded-lg bg-[#FBF9F5] border border-[#EBE7DF] text-xs font-bold text-[#8C8880]">
                        ⏱️ {currentRecipe.cookTime} cocción
                      </span>
                      <span className="px-2.5 py-1 rounded-lg bg-[#FBF9F5] border border-[#EBE7DF] text-xs font-bold text-[#8C8880]">
                        🔥 {currentRecipe.calories}
                      </span>
                    </div>
                    <h2 className="font-serif font-black text-2xl sm:text-4xl text-[#2D2A26] leading-tight">
                      {currentRecipe.title}
                    </h2>
                    <p className="text-sm text-[#8C8880] mt-2 leading-relaxed">
                      {currentRecipe.description}
                    </p>
                  </div>

                  {/* Desglose Nutricional */}
                  <div className="grid grid-cols-4 gap-2 p-3 bg-[#FBF9F5] rounded-2xl border border-[#EBE7DF] text-center">
                    <div>
                      <span className="block text-[10px] font-bold text-[#8C8880] uppercase">Proteínas</span>
                      <span className="font-bold text-xs sm:text-sm text-[#2D2A26]">{currentRecipe.nutrition.proteins}</span>
                    </div>
                    <div>
                      <span className="block text-[10px] font-bold text-[#8C8880] uppercase">Carbos</span>
                      <span className="font-bold text-xs sm:text-sm text-[#2D2A26]">{currentRecipe.nutrition.carbs}</span>
                    </div>
                    <div>
                      <span className="block text-[10px] font-bold text-[#8C8880] uppercase">Grasas</span>
                      <span className="font-bold text-xs sm:text-sm text-[#2D2A26]">{currentRecipe.nutrition.fats}</span>
                    </div>
                    <div>
                      <span className="block text-[10px] font-bold text-[#8C8880] uppercase">Fibra</span>
                      <span className="font-bold text-xs sm:text-sm text-[#2D2A26]">{currentRecipe.nutrition.fiber}</span>
                    </div>
                  </div>

                  {/* Ingredientes */}
                  <div>
                    <h3 className="font-serif font-bold text-lg text-[#2D2A26] mb-3">
                      Ingredientes necesarios ({currentRecipe.servings} raciones)
                    </h3>
                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
                      {currentRecipe.ingredients.map((ing, i) => (
                        <div key={i} className="flex items-center gap-3 p-2.5 rounded-xl bg-[#FBF9F5] border border-[#EBE7DF]">
                          <span className="text-xl">{ing.icon}</span>
                          <div className="text-xs">
                            <span className="font-bold text-[#2D2A26] block">{ing.name}</span>
                            <span className="text-[#8C8880]">{ing.amount}</span>
                          </div>
                        </div>
                      ))}
                    </div>
                  </div>

                  {/* Modo Cocina Paso a Paso */}
                  <div>
                    <div className="flex items-center justify-between mb-3">
                      <h3 className="font-serif font-bold text-lg text-[#2D2A26]">
                        Paso a Paso
                      </h3>
                      <button
                        onClick={() => {
                          setCookingStepIndex(0);
                          const initialMinutes = currentRecipe.steps[0].timerMinutes || 3;
                          setActiveTimer({ secondsLeft: initialMinutes * 60, isRunning: false });
                        }}
                        className="px-3 py-1.5 rounded-xl bg-[#FF6B35] text-white text-xs font-bold flex items-center gap-1.5 shadow-sm hover:bg-[#e05623] cursor-pointer"
                      >
                        ▶️ Iniciar Modo Cocina
                      </button>
                    </div>

                    {/* Lista de Pasos */}
                    <div className="space-y-3">
                      {currentRecipe.steps.map((step, idx) => (
                        <div 
                          key={idx}
                          className={`p-4 rounded-2xl border transition-all ${
                            cookingStepIndex === idx
                              ? "bg-amber-500/5 border-[#FF6B35] shadow-sm"
                              : "bg-white border-[#EBE7DF]"
                          }`}
                        >
                          <div className="flex items-start gap-3">
                            <div className="w-6 h-6 rounded-full bg-[#2D2A26] text-white text-xs font-bold flex items-center justify-center shrink-0 mt-0.5">
                              {idx + 1}
                            </div>
                            <div className="flex-1 space-y-1">
                              <h4 className="font-bold text-xs sm:text-sm text-[#2D2A26]">{step.title}</h4>
                              <p className="text-xs sm:text-sm text-[#8C8880] leading-relaxed">{step.instruction}</p>
                              
                              {/* Temporizador específico del paso */}
                              {cookingStepIndex === idx && (
                                <div className="pt-3 flex items-center gap-3">
                                  <div className="px-3 py-1.5 rounded-xl bg-white border border-[#EBE7DF] font-mono font-bold text-sm text-[#FF6B35]">
                                    ⏱️ {Math.floor((activeTimer?.secondsLeft || 0) / 60)}:
                                    {((activeTimer?.secondsLeft || 0) % 60).toString().padStart(2, '0')}
                                  </div>
                                  <button
                                    onClick={() => setActiveTimer(prev => ({ ...prev, isRunning: !prev.isRunning }))}
                                    className="px-3 py-1.5 rounded-xl bg-[#2D2A26] text-white text-xs font-bold cursor-pointer"
                                  >
                                    {activeTimer?.isRunning ? "Pausar" : "Iniciar Temporizador"}
                                  </button>
                                  {idx < currentRecipe.steps.length - 1 && (
                                    <button
                                      onClick={() => {
                                        const nextIdx = idx + 1;
                                        setCookingStepIndex(nextIdx);
                                        const nextMin = currentRecipe.steps[nextIdx].timerMinutes || 3;
                                        setActiveTimer({ secondsLeft: nextMin * 60, isRunning: false });
                                      }}
                                      className="px-3 py-1.5 rounded-xl bg-[#FF6B35] text-white text-xs font-bold cursor-pointer ml-auto"
                                    >
                                      Siguiente Paso →
                                    </button>
                                  )}
                                </div>
                              )}
                            </div>
                          </div>
                        </div>
                      ))}
                    </div>
                  </div>
                </div>
              </div>
            )}
          </main>

          {/* FOOTER */}
          <footer className="border-t border-[#EBE7DF] bg-white py-6 text-center text-xs text-[#8C8880]">
            <p>ChefIA © 2026 — Asistente Culinario con Inteligencia Artificial. Todos los derechos reservados.</p>
          </footer>

          {/* MODAL DE PAGO (PAYWALL 2,99 €/MES) */}
          {isPaywallOpen && !isAdmin && (
            <div className="fixed inset-0 z-50 bg-[#2D2A26]/75 backdrop-blur-xs flex items-center justify-center p-4">
              <div className="bg-[#FCFAF7] w-full max-w-lg rounded-3xl shadow-2xl border-2 border-[#EBE7DF] overflow-hidden flex flex-col p-6 space-y-5 animate-fade-in">
                {/* Cabecera */}
                <div className="flex items-center justify-between border-b border-[#EBE7DF] pb-4">
                  <div className="flex items-center gap-2.5">
                    <div className="w-8 h-8 rounded-xl bg-[#FF6B35] text-white flex items-center justify-center font-bold">
                      👑
                    </div>
                    <div>
                      <span className="text-[10px] font-bold uppercase tracking-widest text-[#FF6B35]">Límite Diario Alcanzado</span>
                      <h3 className="font-serif font-black text-xl text-[#2D2A26]">Desbloquea ChefIA Pro</h3>
                    </div>
                  </div>
                  <button onClick={() => setIsPaywallOpen(false)} className="text-[#8C8880] hover:text-[#2D2A26] font-bold p-1 cursor-pointer">✕</button>
                </div>

                {/* Banner de Aviso */}
                <div className="p-4 rounded-2xl bg-gradient-to-r from-amber-500/15 via-[#FF6B35]/15 to-amber-500/15 border-2 border-[#FF6B35] text-[#2D2A26]">
                  <p className="font-serif font-black text-base leading-snug">
                    ¡Has alcanzado tu receta gratuita de hoy! Pasa a ChefIA Pro por solo 2,99 €/mes para cocinar sin límites.
                  </p>
                  <p className="text-xs text-[#8C8880] mt-1 font-medium">
                    El contador se reinicia automáticamente cada 24 horas.
                  </p>
                </div>

                {/* Ventajas */}
                <div className="space-y-2">
                  {[
                    "Generación ilimitada de recetas personalizadas",
                    "Modo Cocina Paso a Paso interactivo y temporizadores",
                    "Información nutricional completa y macros",
                    "Guardado ilimitado de recetas favoritas",
                    "Sugerencias de sustitución de ingredientes"
                  ].map((feat, idx) => (
                    <div key={idx} className="flex items-center gap-2 text-xs font-semibold text-[#2D2A26]">
                      <span className="text-emerald-600 font-bold">✓</span>
                      <span>{feat}</span>
                    </div>
                  ))}
                </div>

                {/* Botón Principal Paywall */}
                <div className="pt-2 flex flex-col gap-2">
                  <button
                    onClick={handleUnlockPro}
                    className="w-full py-4 px-6 rounded-2xl bg-gradient-to-r from-[#FF5722] via-[#FF6B35] to-[#FF5722] hover:from-[#E64A19] hover:to-[#E64A19] text-white font-black text-sm shadow-xl shadow-[#FF5722]/30 transition-all uppercase tracking-wide cursor-pointer"
                  >
                    DESBLOQUEAR CHEFIA PRO (2,99 €/MES)
                  </button>
                  <button
                    onClick={() => setIsPaywallOpen(false)}
                    className="py-2.5 text-xs font-bold text-[#8C8880] hover:text-[#2D2A26] transition-colors cursor-pointer"
                  >
                    Cerrar aviso
                  </button>
                </div>
              </div>
            </div>
          )}

          {/* MODAL DIÁLOGO SECRETO ADMIN (3 TOQUES EN LOGO "C") */}
          {showSecretModal && (
            <div 
              className="fixed inset-0 z-50 bg-[#2D2A26]/80 backdrop-blur-xs flex items-center justify-center p-4"
              onClick={() => setShowSecretModal(false)}
            >
              <div 
                className="bg-[#FCFAF7] w-full max-w-sm rounded-3xl shadow-2xl border-2 border-[#EBE7DF] p-6 space-y-4 animate-fade-in"
                onClick={(e) => e.stopPropagation()}
              >
                <div className="flex items-center justify-between">
                  <div className="flex items-center gap-2">
                    <div className="w-8 h-8 rounded-xl bg-[#2D2A26] text-white flex items-center justify-center text-xs">
                      🔒
                    </div>
                    <h3 className="font-serif font-bold text-lg text-[#2D2A26]">Acceso Secreto</h3>
                  </div>
                  <button onClick={() => setShowSecretModal(false)} className="text-[#8C8880] font-bold">✕</button>
                </div>

                <form onSubmit={handleSecretSubmit} className="space-y-4">
                  <div>
                    <label className="block text-xs font-semibold text-[#2D2A26] mb-1.5">
                      Introduce la clave de acceso de administrador:
                    </label>
                    <input
                      type="password"
                      value={secretInputKey}
                      onChange={(e) => { setSecretInputKey(e.target.value); setSecretError(null); }}
                      placeholder="Introduce clave..."
                      className="w-full px-3.5 py-2.5 text-sm bg-white border border-[#EBE7DF] rounded-xl focus:outline-none focus:ring-2 focus:ring-[#FF6B35] font-mono"
                      autoFocus
                    />
                    {secretError && (
                      <p className="text-xs font-bold text-red-600 mt-1.5">⚠️ {secretError}</p>
                    )}
                  </div>

                  <div className="flex items-center gap-2">
                    <button
                      type="button"
                      onClick={() => setShowSecretModal(false)}
                      className="flex-1 py-2.5 rounded-xl bg-[#F2EFE9] text-xs font-bold text-[#8C8880]"
                    >
                      Cancelar
                    </button>
                    <button
                      type="submit"
                      className="flex-1 py-2.5 rounded-xl bg-[#FF6B35] text-white text-xs font-bold shadow-md"
                    >
                      Aceptar
                    </button>
                  </div>
                </form>
              </div>
            </div>
          )}

          {/* TOAST CONFIRMACIÓN MODO ADMIN */}
          {showAdminToast && (
            <div className="fixed top-5 left-1/2 -translate-x-1/2 z-50 bg-[#2D2A26] text-white px-5 py-3 rounded-full shadow-2xl border-2 border-emerald-400 flex items-center gap-2 text-xs sm:text-sm font-bold animate-bounce">
              <span className="text-emerald-400">✓</span>
              <span>¡Modo Administrador Activado!</span>
            </div>
          )}

          {/* DRAWER DE RECETAS FAVORITAS */}
          {isSavedDrawerOpen && (
            <div 
              className="fixed inset-0 z-50 bg-[#2D2A26]/60 backdrop-blur-xs flex justify-end"
              onClick={() => setIsSavedDrawerOpen(false)}
            >
              <div 
                className="bg-[#FCFAF7] w-full max-w-md h-full shadow-2xl border-l border-[#EBE7DF] p-6 flex flex-col justify-between"
                onClick={(e) => e.stopPropagation()}
              >
                <div>
                  <div className="flex items-center justify-between pb-4 border-b border-[#EBE7DF] mb-4">
                    <h3 className="font-serif font-black text-xl text-[#2D2A26]">Recetas Guardadas ({savedRecipes.length})</h3>
                    <button onClick={() => setIsSavedDrawerOpen(false)} className="text-[#8C8880] font-bold p-1">✕</button>
                  </div>

                  {savedRecipes.length === 0 ? (
                    <div className="text-center py-12 text-sm text-[#8C8880]">
                      No tienes recetas guardadas todavía. ¡Genera una y dale a guardar!
                    </div>
                  ) : (
                    <div className="space-y-3 overflow-y-auto max-h-[75vh] custom-scrollbar">
                      {savedRecipes.map((r, i) => (
                        <div 
                          key={i}
                          onClick={() => { setCurrentRecipe(r); setIsSavedDrawerOpen(false); }}
                          className="p-4 rounded-2xl bg-white border border-[#EBE7DF] hover:border-[#FF6B35] cursor-pointer transition-all shadow-xs"
                        >
                          <h4 className="font-serif font-bold text-sm text-[#2D2A26]">{r.title}</h4>
                          <p className="text-xs text-[#8C8880] mt-1 line-clamp-2">{r.description}</p>
                          <div className="flex items-center gap-2 mt-2 text-[10px] font-bold text-[#FF6B35]">
                            <span>⏱️ {r.cookTime}</span>
                            <span>•</span>
                            <span>🔥 {r.calories}</span>
                          </div>
                        </div>
                      ))}
                    </div>
                  )}
                </div>

                <button
                  onClick={() => setIsSavedDrawerOpen(false)}
                  className="w-full py-3 rounded-xl bg-[#2D2A26] text-white text-xs font-bold"
                >
                  Cerrar
                </button>
              </div>
            </div>
          )}
        </div>
      );
    }

    // Montar la aplicación
    ReactDOM.render(<App />, document.getElementById('root'));
  </script>
</body>
